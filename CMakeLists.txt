# AcePS build definition: configures portable emulator foundations and the optional Qt desktop shell.
cmake_minimum_required(VERSION 3.21)
if(NOT DEFINED CMAKE_TOOLCHAIN_FILE AND EXISTS "${CMAKE_SOURCE_DIR}/.tools/vcpkg/scripts/buildsystems/vcpkg.cmake")
    set(CMAKE_TOOLCHAIN_FILE "${CMAKE_SOURCE_DIR}/.tools/vcpkg/scripts/buildsystems/vcpkg.cmake" CACHE FILEPATH "vcpkg toolchain")
endif()
project(AcePS VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
add_library(aceps_engine
    support/logging/logger.cpp engine/memory_space/address_space.cpp engine/processor/jaguar_cpu.cpp
    engine/kernel/kernel_services.cpp engine/content/virtual_file_system.cpp engine/content/package_reader.cpp
    engine/graphics/graphics_device.cpp engine/sound/audio_output.cpp engine/controls/input_router.cpp
    runtime/emulation_session.cpp)
target_include_directories(aceps_engine PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(aceps_engine PUBLIC cxx_std_20)
if(MSVC)
    target_compile_options(aceps_engine PRIVATE /W4 /permissive-)
else()
    target_compile_options(aceps_engine PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(aceps_runtime_smoke tests/runtime_smoke.cpp)
target_link_libraries(aceps_runtime_smoke PRIVATE aceps_engine)
add_executable(aceps_processor_tests tests/processor_tests.cpp)
target_link_libraries(aceps_processor_tests PRIVATE aceps_engine)
add_executable(aceps_address_space_tests tests/address_space_tests.cpp)
target_link_libraries(aceps_address_space_tests PRIVATE aceps_engine)
if(BUILD_TESTING)
    add_test(NAME runtime_smoke COMMAND aceps_runtime_smoke)
    add_test(NAME processor_tests COMMAND aceps_processor_tests)
    add_test(NAME address_space_tests COMMAND aceps_address_space_tests)
endif()
option(ACEPS_BUILD_DESKTOP "Build the Qt6 desktop interface" ON)
if(ACEPS_BUILD_DESKTOP)
    find_package(Qt6 6.5 QUIET COMPONENTS Widgets)
    if(Qt6Widgets_FOUND)
        set(CMAKE_AUTOMOC ON)
        add_executable(AcePS app/desktop/main.cpp interface/shell/main_window.cpp interface/shell/main_window.h)
        target_link_libraries(AcePS PRIVATE aceps_engine Qt6::Widgets)
    else()
        message(WARNING "Qt6 Widgets was not found; only the core runtime smoke target will be built. Run setup.sh or setup.bat to install it.")
    endif()
endif()
